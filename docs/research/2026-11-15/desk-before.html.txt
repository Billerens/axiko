<script>
  const page = document.getElementById('page');
  const chat = document.getElementById('chat');
  const open = document.getElementById('open');
  const code = document.getElementById('code');

  // ---- the phone: the same calls its own pages make through their bridge ------------------------

  async function rpc(body) {
    const answer = await fetch('/__assi/rpc', { method: 'POST', body: JSON.stringify(body) }).then(response => response.json());
    if (answer.error) throw new Error(answer.error);
    return answer.result === undefined ? undefined : JSON.parse(answer.result);
  }
  const native = (method, ...args) => rpc({ op: 'native', method, args: JSON.stringify(args) });
  const engine = source => rpc({ op: 'eval', code: source });
  const files = (fn, ...args) => rpc({ op: 'call', path: 'lib/assi.js', fn: 'files.' + fn, args: JSON.stringify(args) })
    .catch(() => engine('require("lib/assi").files.' + fn + '(' + args.map(arg => JSON.stringify(arg)).join(', ') + ')'));

  // ---- views ------------------------------------------------------------------------------------

  function view(name) {
    document.querySelectorAll('#bar button').forEach(button => button.classList.toggle('on', button.dataset.view === name));
    document.querySelectorAll('.view').forEach(node => node.classList.toggle('on', node.id === name));
    localStorage.setItem('desk.view', name);
    if (name === 'files' && !document.getElementById('tree').children.length) list('', document.getElementById('tree'), 0);
    if (name === 'journal') journal();
  }
  document.querySelectorAll('#bar button').forEach(button => { button.onclick = () => view(button.dataset.view); });

  function showChat(visible) {
    chat.hidden = !visible;
    open.hidden = visible;
    localStorage.setItem('desk.chat', visible ? 'on' : 'off');
  }

  // What the frames call instead of the phone's shell.
  window.assiDesk = {
    open: path => { page.src = '/' + String(path).replace(/^\//, ''); view('screen'); },
    back: () => { page.contentWindow.history.back(); },
    chat: (visible, agent) => {
      showChat(visible);
      if (agent && chat.contentWindow.__assiEmit) chat.contentWindow.__assiEmit('chat.agent', agent);
    },
    where: () => {
      try { return page.contentWindow.location.pathname.replace(/^\//, ''); } catch (error) { return ''; }
    },
  };
  page.addEventListener('load', () => { document.getElementById('where').textContent = window.assiDesk.where(); });

  // ---- files --------------------------------------------------------------------------------------

  let current = null;
  let saved = '';

  async function list(path, into, depth) {
    const names = await engine('require("lib/assi").files.list(' + JSON.stringify(path) + ')');
    names.sort((a, b) => (b.endsWith('/') - a.endsWith('/')) || a.localeCompare(b));
    for (const name of names) {
      const dir = name.endsWith('/');
      const full = path + name;
      const row = document.createElement('div');
      row.className = 'item' + (dir ? ' dir' : '');
      row.style.paddingLeft = (10 + depth * 14) + 'px';
      row.textContent = (dir ? '▸ ' : '  ') + name;
      into.append(row);
      if (dir) {
        const box = document.createElement('div');
        box.hidden = true;
        into.append(box);
        row.onclick = async () => {
          box.hidden = !box.hidden;
          row.textContent = (box.hidden ? '▸ ' : '▾ ') + name;
          if (!box.hidden && !box.children.length) await list(full, box, depth + 1);
        };
      } else {
        row.onclick = () => edit(full, row);
      }
    }
  }

  async function edit(path, row) {
    if (current && code.value !== saved && !confirm('В файле ' + current + ' есть несохранённые правки. Бросить их?')) return;
    document.querySelectorAll('#tree .item.on').forEach(node => node.classList.remove('on'));
    if (row) row.classList.add('on');
    current = path;
    document.getElementById('file').textContent = path;
    code.disabled = true;
    code.value = '';
    try {
      saved = await engine('require("lib/assi").files.read(' + JSON.stringify(path) + ')');
      code.value = saved;
      code.disabled = false;
    } catch (error) {
      code.value = 'Не открывается как текст: ' + error.message;
    }
    document.getElementById('save').disabled = true;
    document.getElementById('show').disabled = !/\.html?$/.test(path);
  }

  async function save() {
    if (!current || code.disabled) return;
    await engine('require("lib/assi").files.write(' + JSON.stringify(current) + ', ' + JSON.stringify(code.value) + ')');
    saved = code.value;
    document.getElementById('save').disabled = true;
    // A screen that is showing this file is drawn anew.
    if (window.assiDesk.where() === current) page.contentWindow.location.reload();
  }

  code.addEventListener('input', () => { document.getElementById('save').disabled = code.value === saved; });
  code.addEventListener('keydown', event => {
    if ((event.ctrlKey || event.metaKey) && event.key === 's') { event.preventDefault(); save(); }
    if (event.key === 'Tab') {
      event.preventDefault();
      const at = code.selectionStart;
      code.setRangeText('  ', at, code.selectionEnd, 'end');
      code.dispatchEvent(new Event('input'));
    }
  });
  document.getElementById('save').onclick = save;
  document.getElementById('show').onclick = () => window.assiDesk.open(current);
  window.addEventListener('beforeunload', event => { if (current && code.value !== saved) event.preventDefault(); });

  // ---- the journal --------------------------------------------------------------------------------

  async function journal() {
    if (!document.getElementById('journal').classList.contains('on')) return;
    const log = document.getElementById('log');
    const pinned = log.scrollHeight - log.scrollTop - log.clientHeight < 60;
    const lines = await native('log.engine', 400).catch(error => ['Журнал недоступен: ' + error.message]);
    log.textContent = lines.length ? lines.join('\n') : 'Журнал движка пуст.';
    if (pinned) log.scrollTop = log.scrollHeight;
  }
  setInterval(journal, 3000);

  open.onclick = () => showChat(true);
  showChat(localStorage.getItem('desk.chat') !== 'off');
  view(localStorage.getItem('desk.view') || 'screen');

  // When an agent opens a screen to check its work, show the same screen here; a file it
  // changed while it is open in the editor is taken anew, unless there are unsaved edits.
  const events = new EventSource('/__assi/events');
  events.onmessage = message => {
    const event = JSON.parse(message.data);
    if (event.event === 'ui.opened') window.assiDesk.open(event.data);
    if (event.event === 'agent' && event.data.type === 'changes' && current && code.value === saved) edit(current);
  };
</script>
